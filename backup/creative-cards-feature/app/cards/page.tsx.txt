import type { Metadata } from "next";
import Link from "next/link";
import { cookies } from "next/headers";
import { CardsGallery } from "@/components/cards-gallery";
import { CREATIVE_CARD_COOKIE, CreativeCardsError, getCreativeCardsSnapshot } from "@/lib/creative-cards";
import type { CreativeCard } from "@/lib/creative-cards";

export const metadata: Metadata = {
  title: "Creative Cards | OnlyPaws",
  description: "A simple download gallery for the finished dachshund creative cards.",
};

export default async function CardsPage() {
  const cookieStore = await cookies();
  const voterToken = cookieStore.get(CREATIVE_CARD_COOKIE)?.value ?? null;
  let cards: CreativeCard[] = [];
  let unavailableMessage: string | null = null;

  try {
    cards = await getCreativeCardsSnapshot(voterToken);
  } catch (error) {
    console.error("[cards] page_load_failed", error);
    unavailableMessage =
      error instanceof CreativeCardsError && error.code === "schema_missing"
        ? "The card gallery is almost ready. The voting tables still need to be created in Supabase."
        : "The card gallery is temporarily unavailable right now. Please try again in a little while.";
  }

  return (
    <main className="shell shellSimple">
      <section className="cardsPage">
        <header className="cardsHeader">
          <div className="cardsHeaderTop">
            <Link className="brandMark brandMarkInline" href="/">
              <span className="brandBadge">OP</span>
              <span className="brandText">OnlyPaws</span>
            </Link>

            <Link className="navLink cardsBackLink" href="/submit">
              Back to poll
            </Link>
          </div>

          <div className="cardsIntro">
            <p className="sectionEyebrow cardsEyebrow">Creative work</p>
            <h1 className="simpleTitle cardsTitle">Finished cards, ready to download.</h1>
            <p className="introText cardsBody">
              A simple gallery of the custom dog cards. Tap any download button to save the image directly.
            </p>
          </div>
        </header>

        {unavailableMessage ? (
          <div className="panel cardsUnavailable">
            <h2 className="cardsUnavailableTitle">Gallery unavailable</h2>
            <p className="cardsUnavailableText">{unavailableMessage}</p>
          </div>
        ) : (
          <CardsGallery initialCards={cards} />
        )}
      </section>
    </main>
  );
}
